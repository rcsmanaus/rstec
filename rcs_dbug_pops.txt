#include <WiFi.h>
#include <PubSubClient.h>
#include <DHT.h>

// ============ CONFIGURAÇÕES ============
#define WIFI_SSID "2PISO"
#define WIFI_PASSWORD "Ric@rd0s"
#define MQTT_SERVER "193.168.1.109"

#define DHTPIN 4
#define DHTTYPE DHT22
#define SMOKE_SENSOR_PIN 34
#define ACS712_PIN 35

#define RELAY1 16
#define RELAY2 17
#define RELAY3 18
#define RELAY4 19

// IP fixo
IPAddress local_IP(193, 168, 1, 130);
IPAddress gateway(193, 168, 1, 100);
IPAddress subnet(255, 255, 255, 0);

DHT dht(DHTPIN, DHTTYPE);
WiFiClient espClient;
PubSubClient client(espClient);

unsigned long lastMsg = 0;
const long interval = 5000;

// ============ SETUP ============
void setup() {
  Serial.begin(115200);
  dht.begin();

  pinMode(SMOKE_SENSOR_PIN, INPUT);
  pinMode(ACS712_PIN, INPUT);

  pinMode(RELAY1, OUTPUT);
  pinMode(RELAY2, OUTPUT);
  pinMode(RELAY3, OUTPUT);
  pinMode(RELAY4, OUTPUT);

  digitalWrite(RELAY1, HIGH);
  digitalWrite(RELAY2, HIGH);
  digitalWrite(RELAY3, HIGH);
  digitalWrite(RELAY4, HIGH);

  setup_wifi();
  client.setServer(MQTT_SERVER, 1883);
  client.setCallback(callback);
}

void setup_wifi() {
  delay(10);
  Serial.println("Conectando ao Wi-Fi...");

  if (!WiFi.config(local_IP, gateway, subnet)) {
    Serial.println("❌ Falha ao configurar IP fixo");
  }

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\n✅ Wi-Fi conectado!");
  Serial.print("Endereço IP: ");
  Serial.println(WiFi.localIP());
}

// ============ MQTT CALLBACK ============
void callback(char* topic, byte* payload, unsigned int length) {
  String msg;
  for (int i = 0; i < length; i++) {
    msg += (char)payload[i];
  }

  Serial.print("Comando recebido [");
  Serial.print(topic);
  Serial.print("] ");
  Serial.println(msg);

  if (String(topic) == "casa/sala/rele1") digitalWrite(RELAY1, (msg == "ON") ? LOW : HIGH);
  if (String(topic) == "casa/sala/rele2") digitalWrite(RELAY2, (msg == "ON") ? LOW : HIGH);
  if (String(topic) == "casa/sala/rele3") digitalWrite(RELAY3, (msg == "ON") ? LOW : HIGH);
  if (String(topic) == "casa/sala/rele4") digitalWrite(RELAY4, (msg == "ON") ? LOW : HIGH);
}

// ============ RECONNECT MQTT ============
void reconnect() {
  while (!client.connected()) {
    Serial.print("Tentando conectar ao MQTT...");
    if (client.connect("ESP32Client")) {
      Serial.println("Conectado!");
      client.subscribe("casa/sala/rele1");
      client.subscribe("casa/sala/rele2");
      client.subscribe("casa/sala/rele3");
      client.subscribe("casa/sala/rele4");
    } else {
      Serial.print("Falha, rc=");
      Serial.print(client.state());
      Serial.println(" tentando novamente em 5s");
      delay(5000);
    }
  }
}

// ============ LEITURA DA CORRENTE EM AMPÉRES ============
float lerCorrenteACS712() {
  const int numLeituras = 1000;
  float media = 0;

  for (int i = 0; i < numLeituras; i++) {
    int leitura = analogRead(ACS712_PIN);
    float tensao = (leitura * 3.3) / 4095.0;   // Converter para tensão
    float correnteInst = (tensao - 2.5);       // Centro do sensor é 2.5V
    correnteInst = correnteInst / 0.1;         // 100 mV por ampere (ACS712-20A)
    media += correnteInst * correnteInst;      // quadrado da corrente
  }

  float rms = sqrt(media / numLeituras);       // Corrente RMS
  return rms;
}

// ============ LOOP ============
void loop() {
  if (!client.connected()) {
    reconnect();
  }
  client.loop();

  unsigned long now = millis();
  if (now - lastMsg > interval) {
    lastMsg = now;

    float t = dht.readTemperature();
    float h = dht.readHumidity();
    int smoke = analogRead(SMOKE_SENSOR_PIN);
    float corrente = lerCorrenteACS712();

    if (!isnan(t)) client.publish("casa/sala/temperatura", String(t).c_str());
    if (!isnan(h)) client.publish("casa/sala/umidade", String(h).c_str());

    client.publish("casa/sala/fumaca", String(smoke).c_str());
    client.publish("casa/sala/corrente", String(corrente, 2).c_str());  // corrente em Amperes
  }
}
